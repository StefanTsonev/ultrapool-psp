cmake_minimum_required(VERSION 3.16)
project(ultrapool_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ultrapool_core src/core/menu_features.cpp src/core/shop_stock.cpp src/core/daily.cpp src/core/daily_history.cpp src/core/daily_game.cpp src/core/progression.cpp src/core/progression_catalog.cpp src/core/progress_events.cpp src/core/remaining_effects.cpp src/core/identity_effects.cpp src/core/passive_balls.cpp src/core/rituals.cpp src/core/collection.cpp src/core/pocket_events.cpp src/core/summons.cpp src/core/holes.cpp src/core/movement.cpp src/core/projectiles.cpp src/core/content_effects.cpp src/core/effects.cpp src/core/shop_events.cpp src/core/decks.cpp src/core/cubes.cpp src/core/audio.cpp src/core/pickups.cpp src/core/passives.cpp src/core/splitting.cpp src/core/ball_events.cpp src/core/modifiers.cpp src/core/buffs.cpp src/core/catalog.cpp src/core/listeners.cpp src/core/mixing.cpp src/core/actions.cpp src/core/save.cpp src/core/prediction.cpp src/core/physics.cpp src/core/game.cpp src/core/resources.cpp src/core/content.cpp src/core/presentation.cpp)
target_include_directories(ultrapool_core PUBLIC include)
if(WIN32)
  target_compile_definitions(ultrapool_core PUBLIC _CRT_SECURE_NO_WARNINGS)
endif()
if(MSVC)
  target_compile_options(ultrapool_core PRIVATE /W4)
else()
  target_compile_options(ultrapool_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(core_tests tests/menu_features_tests.cpp tests/shop_transform_tests.cpp tests/daily_tests.cpp tests/progression_tests.cpp tests/score_contact_tests.cpp tests/identity_effect_tests.cpp tests/passive_body_tests.cpp tests/ritual_tests.cpp tests/collection_tests.cpp tests/summon_tests.cpp tests/hole_tests.cpp tests/movement_tests.cpp tests/projectile_tests.cpp tests/content_batch_tests.cpp tests/core_tests.cpp tests/mixing_tests.cpp tests/delayed_effect_tests.cpp tests/listener_tests.cpp tests/buff_tests.cpp tests/contact_tests.cpp tests/modifier_tests.cpp tests/fire_tests.cpp tests/split_tests.cpp tests/passive_tests.cpp tests/pickup_tests.cpp tests/audio_tests.cpp tests/ui_refresh_tests.cpp tests/audit_fix_tests.cpp)
target_link_libraries(core_tests PRIVATE ultrapool_core)
add_test(NAME core_tests COMMAND core_tests)
set_tests_properties(core_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32)
  add_executable(ultrapool_host WIN32 src/host/windows.cpp)
  target_link_libraries(ultrapool_host PRIVATE ultrapool_core user32 gdi32 msimg32)
endif()
add_executable(visual_audit tools/visual_audit.cpp)
target_link_libraries(visual_audit PRIVATE ultrapool_core)
